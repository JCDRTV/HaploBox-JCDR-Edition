-- HaploBox Profession Guide – catalog of profession identities (not a leveling recipe DB).
-- SkillLine IDs are stable; skill values are read from the active game client.
HaploBoxProfessionGuide = HaploBoxProfessionGuide or {}
local PG = HaploBoxProfessionGuide

PG.VERSION = "HF26.PG.1-TEST"
PG.PREFIX = "HBPG1"
PG.ORDER = {"BS", "LW", "ENG", "TAIL", "ALCH", "ENCH", "MIN", "HERB", "SKIN"}
PG.PROFESSIONS = {
    BS   = { name = "Blacksmithing", id = 164, icon = "Interface\\Icons\\Trade_BlackSmithing" },
    LW   = { name = "Leatherworking", id = 165, icon = "Interface\\Icons\\Trade_LeatherWorking" },
    ENG  = { name = "Engineering", id = 202, icon = "Interface\\Icons\\Trade_Engineering" },
    TAIL = { name = "Tailoring", id = 197, icon = "Interface\\Icons\\Trade_Tailoring" },
    ALCH = { name = "Alchemy", id = 171, icon = "Interface\\Icons\\Trade_Alchemy" },
    ENCH = { name = "Enchanting", id = 333, icon = "Interface\\Icons\\Trade_Engraving" },
    MIN  = { name = "Mining", id = 186, icon = "Interface\\Icons\\Trade_Mining" },
    HERB = { name = "Herbalism", id = 182, icon = "Interface\\Icons\\Trade_Herbalism" },
    SKIN = { name = "Skinning", id = 393, icon = "Interface\\Icons\\INV_Misc_Pelt_Wolf_01" },
}
-- Localization-safe aliases can be extended without changing any saved records.
PG.BY_NAME = {
    ["blacksmithing"]="BS", ["leatherworking"]="LW", ["engineering"]="ENG",
    ["tailoring"]="TAIL", ["alchemy"]="ALCH", ["enchanting"]="ENCH",
    ["mining"]="MIN", ["herbalism"]="HERB", ["skinning"]="SKIN",
}
PG.BY_ID = {}
for code, item in pairs(PG.PROFESSIONS) do PG.BY_ID[item.id] = code end
PG.BANDS = {
    { startSkill=1, endSkill=75, name="Apprentice" },
    { startSkill=75, endSkill=150, name="Journeyman" },
    { startSkill=150, endSkill=225, name="Expert" },
    { startSkill=225, endSkill=300, name="Artisan" },
}

-- Real recipe steps are registered only after source and in-game validation.
-- Shape: { { from=1, to=30, name="...", spellID=123, reagents={{itemID=1,count=2}} } }
PG.GUIDES = {}
function PG.RegisterGuide(code, rows)
    if not PG.PROFESSIONS[code] or type(rows) ~= "table" then return false end
    for _, row in ipairs(rows) do
        if type(row.from) ~= "number" or type(row.to) ~= "number"
            or row.from < 1 or row.to > 300 or row.from >= row.to
            or type(row.name) ~= "string" then return false end
    end
    PG.GUIDES[code] = rows
    if PG.RefreshUI then PG.RefreshUI() end
    return true
end


=== HaploBox/ProfessionGuide/Core.lua ===
-- HaploBox Profession Guide – Phase 1: current-skill scan and party sync.
local PG = HaploBoxProfessionGuide
local function now()
    return type(GetServerTime) == "function" and GetServerTime() or (type(time) == "function" and time() or 0)
end
local function clock()
    return type(GetTime) == "function" and GetTime() or 0
end

function PG.ShortName(name)
    if type(name) ~= "string" then return nil end
    local n = name:match("^([^%-]+)")
    if not n or n == "" then return nil end
    return n
end
function PG.Me()
    return PG.ShortName(UnitName("player")) or "Player"
end

local function findCode(name, skillLineID)
    local code = PG.BY_ID[tonumber(skillLineID) or 0]
    if code then return code end
    if type(name) ~= "string" then return nil end
    local normalized = name:lower():gsub("^%s+", ""):gsub("%s+$", "")
    if PG.BY_NAME[normalized] then return PG.BY_NAME[normalized] end
    -- Classic/Forever may prefix a profession with a skill tier/expansion name.
    for professionName, match in pairs(PG.BY_NAME) do
        if normalized:match("%f[%a]" .. professionName .. "%f[%A]") then return match end
    end
end

function PG.ReadMySkills()
    local results = {}
    -- A modern API exists on Forever; the legacy fallback is useful on Classic Era.
    if type(GetProfessions) == "function" and type(GetProfessionInfo) == "function" then
        local indexes = { GetProfessions() }
        for _, index in pairs(indexes) do
            if type(index) == "number" then
                local ok, name, _, skill, maxSkill, _, _, skillLine = pcall(GetProfessionInfo, index)
                if ok then
                    local code = findCode(name, skillLine)
                    if code and type(skill) == "number" then
                        results[code] = {skill=math.max(0, math.floor(skill)), cap=math.max(0, math.floor(tonumber(maxSkill) or 0))}
                    end
                end
            end
        end
    end
    -- Forever's C_TradeSkillUI can report professions that GetProfessions omits.
    local trade = C_TradeSkillUI
    if trade and type(trade.GetProfessionInfoBySkillLineID) == "function" then
        for _, code in ipairs(PG.ORDER) do
            local id = PG.PROFESSIONS[code].id
            local ok, info = pcall(trade.GetProfessionInfoBySkillLineID, id)
            if ok and type(info) == "table" and type(info.skillLevel) == "number" and info.skillLevel > 0 then
                local newSkill = math.floor(info.skillLevel)
                local old = results[code]
                if not old or newSkill > old.skill then
                    results[code] = {skill=newSkill,cap=math.floor(tonumber(info.maxSkillLevel) or 0)}
                end
            end
        end
    end
    local numSkills = C_SkillInfo and C_SkillInfo.GetNumSkillLines or GetNumSkillLines
    local getSkill = C_SkillInfo and C_SkillInfo.GetSkillLineInfo or GetSkillLineInfo
    if type(numSkills) == "function" and type(getSkill) == "function" then
        local ok, total = pcall(numSkills)
        if ok and type(total) == "number" then
            for i=1,math.min(total,500) do
                local valid, first, isHeader, _, rank, _, _, maxRank = pcall(getSkill, i)
                local name
                if valid and type(first) == "table" then
                    name = first.name
                    isHeader = first.isHeader
                    rank = first.rank
                    maxRank = first.maxRank
                else
                    name = first
                end
                if valid and not isHeader and type(rank) == "number" then
                    local code = findCode(name)
                    if code and (not results[code] or rank > results[code].skill) then
                        results[code] = {skill=math.max(0,math.floor(rank)), cap=math.max(0,math.floor(tonumber(maxRank) or 0))}
                    end
                end
            end
        end
    end
    return results
end

function PG.GetRoster()
    local names, seen = {}, {}
    local function add(name)
        name = PG.ShortName(name)
        if name and not seen[name:lower()] then
            seen[name:lower()] = true
            names[#names+1] = name
        end
    end
    add(PG.Me())
    if type(IsInRaid) == "function" and IsInRaid() then
        local total = type(GetNumGroupMembers) == "function" and GetNumGroupMembers() or 0
        for i=1,math.min(total,40) do add(UnitName("raid" .. i)) end
    elseif type(IsInGroup) == "function" and IsInGroup() then
        local total = type(GetNumSubgroupMembers) == "function" and GetNumSubgroupMembers() or 4
        for i=1,math.min(total,4) do add(UnitName("party" .. i)) end
    elseif PG.db and PG.db.team then
        -- When alone, retain the last seen party so the overview remains useful.
        for name in pairs(PG.db.team) do add(name) end
    end
    return names
end

local function inCurrentParty(name)
    if not name then return false end
    for _, current in ipairs(PG.GetRoster()) do
        if name:lower() == current:lower() then return true end
    end
    return false
end

function PG.ScanMySkills()
    if not PG.db then return end
    local me = PG.Me()
    local data = PG.ReadMySkills()
    PG.db.team[me] = {skills=data, updated=now(), online=true}
    if PG.RefreshUI then PG.RefreshUI() end
end

local function transport()
    if type(IsInGroup) ~= "function" or not IsInGroup() then return nil end
    if type(IsInGroup) == "function" and type(LE_PARTY_CATEGORY_INSTANCE) == "number" and IsInGroup(LE_PARTY_CATEGORY_INSTANCE) then return "INSTANCE_CHAT" end
    if type(IsInRaid) == "function" and IsInRaid() then return "RAID" end
    return "PARTY"
end
local function send(message)
    local channel = transport()
    if not channel then return end
    if C_ChatInfo and type(C_ChatInfo.SendAddonMessage) == "function" then
        pcall(C_ChatInfo.SendAddonMessage, PG.PREFIX, message, channel)
    elseif type(SendAddonMessage) == "function" then
        pcall(SendAddonMessage, PG.PREFIX, message, channel)
    end
end
function PG.Broadcast(force)
    if not PG.db then return end
    if not force and clock() - (PG.lastSent or -100) < 1 then return end
    PG.lastSent = clock()
    local me = PG.db.team[PG.Me()]
    if not me then PG.ScanMySkills(); me = PG.db.team[PG.Me()] end
    if not me then return end
    local parts = {}
    for _, code in ipairs(PG.ORDER) do
        local s = me.skills[code]
        if s then parts[#parts+1] = code .. ":" .. s.skill .. ":" .. s.cap end
    end
    send("S|" .. table.concat(parts, ","))
end
local function readSnapshot(name, msg)
    local skills = {}
    local payload = msg:sub(3)
    if #payload > 250 then return end
    for code, skill, cap in payload:gmatch("([A-Z]+):(%d+):(%d+)") do
        if PG.PROFESSIONS[code] then
            local n, m = tonumber(skill), tonumber(cap)
            if n and m and n <= 10000 and m <= 10000 then
                skills[code] = {skill=n,cap=m}
            end
        end
    end
    PG.db.team[name] = {skills=skills,updated=now(),online=true}
    if PG.RefreshUI then PG.RefreshUI() end
end

local events = CreateFrame("Frame")
PG.events = events
events:RegisterEvent("PLAYER_LOGIN")
events:RegisterEvent("PLAYER_ENTERING_WORLD")
events:RegisterEvent("GROUP_ROSTER_UPDATE")
events:RegisterEvent("CHAT_MSG_ADDON")
events:RegisterEvent("SKILL_LINES_CHANGED")
events:RegisterEvent("TRADE_SKILL_SHOW")
events:RegisterEvent("TRADE_SKILL_CLOSE")
events:SetScript("OnEvent", function(_, event, ...)
    if event == "PLAYER_LOGIN" then
        HaploBoxProfessionDB = HaploBoxProfessionDB or {}
        PG.db = HaploBoxProfessionDB
        PG.db.team = type(PG.db.team) == "table" and PG.db.team or {}
        PG.db.selected = PG.PROFESSIONS[PG.db.selected] and PG.db.selected or "LW"
        if C_ChatInfo and type(C_ChatInfo.RegisterAddonMessagePrefix) == "function" then
            pcall(C_ChatInfo.RegisterAddonMessagePrefix, PG.PREFIX)
        elseif type(RegisterAddonMessagePrefix) == "function" then
            pcall(RegisterAddonMessagePrefix, PG.PREFIX)
        end
        PG.ScanMySkills()
        if PG.CreateUI then PG.CreateUI() end
        if C_Timer and type(C_Timer.After) == "function" then
            C_Timer.After(2, function() PG.ScanMySkills(); PG.Broadcast(true); send("R|") end)
        end
        -- A low-frequency heartbeat keeps the party's skill cache honest.
        if C_Timer and type(C_Timer.NewTicker) == "function" then
            PG.heartbeat = C_Timer.NewTicker(60, function()
                if PG.db then PG.ScanMySkills(); PG.Broadcast(true) end
            end)
        end
    elseif not PG.db then
        return
    elseif event == "CHAT_MSG_ADDON" then
        local prefix, msg, _, sender = ...
        if prefix ~= PG.PREFIX or type(msg) ~= "string" or #msg > 250 then return end
        local who = PG.ShortName(sender)
        if not who or who:lower() == PG.Me():lower() or not inCurrentParty(who) then return end
        if msg == "R|" then
            PG.Broadcast(true)
        elseif msg:sub(1,2) == "S|" then
            readSnapshot(who,msg)
        end
    elseif event == "GROUP_ROSTER_UPDATE" or event == "PLAYER_ENTERING_WORLD" then
        PG.ScanMySkills()
        PG.Broadcast(true)
        send("R|")
    else
        PG.ScanMySkills()
        PG.Broadcast(false)
    end
end)

local function fill(parent, color)
    local t=parent:CreateTexture(nil,"BACKGROUND")
    t:SetTexture(WHITE)
    t:SetVertexColor(unpack(color))
    t:SetAllPoints()

end
local function text(parent, textValue, size, color, width, justify)
    local t=parent:CreateFontString(nil,"OVERLAY", size == "large" and "GameFontNormalLarge" or "GameFontHighlight")
    t:SetText(textValue or "")
    t:SetTextColor(unpack(color or C.white))

local function button(parent, label, width, height, callback)

    b:SetScript("OnLeave",function(self) self.tint:SetVertexColor(unpack(C.tile)) end)

    frame:SetMovable(true)

local function progress(parent,w,h)
    local bg=CreateFrame("Frame",nil,parent)
    bg:SetSize(w,h)
    fill(bg,C.tile)
    local bar=bg:CreateTexture(nil,"ARTWORK")
    bar:SetTexture(WHITE)
    bar:SetVertexColor(unpack(C.green))
    bar:SetPoint("LEFT",bg,"LEFT",0,0)
    bar:SetHeight(h)
    function bg:SetPercent(n)
        bar:SetWidth(math.max(0.01,w*math.max(0,math.min(1,n or 0))))
    end
    bg:SetPercent(0)

local function teamSnapshot(code)
    local rows = {}
    local names=PG.GetRoster()
    for _, name in ipairs(names) do
        local state=PG.db and PG.db.team and PG.db.team[name]
        local skill=state and state.skills and state.skills[code]
        rows[#rows+1]={name=name,state=state,skill=skill}
    end
    return rows
end
local function timeStatus(state)
    if not state or not state.updated then return "NO DATA",C.gold end
    local now=type(GetServerTime)=="function" and GetServerTime() or (type(time)=="function" and time() or 0)
    if now>0 and state.updated>0 and (now-state.updated)>180 then return "CACHED",C.gold end
    return "SYNCED",C.green
end
local function selectedSkill(code)
    local rows=teamSnapshot(code)
    local focus=PG.focus
    for _, row in ipairs(rows) do
        if row.name==focus then return row.skill,row.name end
    end
    for _, row in ipairs(rows) do
        if row.skill then
            PG.focus=row.name
            return row.skill,row.name
        end
    end
    PG.focus=rows[1] and rows[1].name or nil
    return nil,PG.focus
end
local function currentBand(skill)
    if not skill then return "Unknown" end
    for _, b in ipairs(PG.BANDS) do
        if skill < b.endSkill then return b.name .. " (" .. b.startSkill .. "-" .. b.endSkill .. ")" end
    end
    return "300 / 300 complete"
end

function PG.RefreshUI()
    if not PG.main or not PG.db then return end
    local code=PG.db.selected or "LW"
    local profession=PG.PROFESSIONS[code]
    PG.mainHead:SetText(profession.name:upper())
    local roster=PG.GetRoster()
    PG.mainDetail:SetText(#roster .. " party character(s)  |  Select a row to focus")
    for _, pcode in ipairs(PG.ORDER) do
        local card=PG.cards[pcode]
        local n, highest=0,0
        for _, row in ipairs(teamSnapshot(pcode)) do
            if row.skill then n=n+1;highest=math.max(highest,row.skill.skill) end
        end
        card.stat:SetText(n==0 and "no skill reported" or (n .. " toon(s)  |  " .. highest .. "/300"))
        if pcode==code then card.back:SetVertexColor(0.17,0.28,0.26,1)
        else card.back:SetVertexColor(unpack(C.tile)) end
